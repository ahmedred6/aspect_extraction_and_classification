library IEEE;
use IEEE.std_logic_1164.all;

entity PBWC is
    port (
        CLock, Reset       : in  std_logic;
        Press              : in  std_logic;
        Open_CW, Close_CCW : out std_logic;
    )
end entity

type state_type is (w_closed, w_open);
signal_current_state, next_state : State_Type;

state_memory : process(clock,reset)
    begin
        if(reset='0') then
            current_state<=w_closed;
        elsif(clock'event and clock='1') then
            current_state<=next_state;
        end if;
    end process


next_state : process(current_state, press)
    begin
        case(current_state) is
            when w_closed=> if (press='0') then
                                next_state<=w_closed;
                            else
                                next_state<=w_open;
                            end if;
            when w_open=> if (press='0') then
                                next_state<=w_open;
                            else
                                next_state<=w_closed;
                            end if;
            when others=>next_state<=w_closed;
        end case;
    end process;

output : process(current_state, press)
    begin
        case(current_state) is
            when w_closed=>if(press='0') then
                            open_cw<='0';close_ccw<='0';
                           else
                            open_cw<='0';close_ccw<='0';
                           end if;
            when w_open=>if(press='0') then
                            open_cw<='0';close_ccw<='1';
                         else 
                             open_cw<='0';close_ccw<='1';
                          end if;
            when others=>open_cw<='0';close_ccw<='0';
        end case;
    end process
